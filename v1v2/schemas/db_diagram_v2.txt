-- Implementação: Trigger para calcular automaticamente total_quantity como soma acumulada de variation
-- Implementado: 2025-12-26 por Danilo Silva
-- Alterado: 2026-03-18 por Lucas Pratas
-- Descrição: Este trigger recalcula total_quantity para todos os registros do mesmo grupo
--            (holder_id, asset_id, lot_id, financial_account_id) sempre que uma nova posição é inserida
--            ou atualizado ou excluída.
--            Os registros são ordenados por date ASC, depois id ASC para o cálculo cumulativo.

-- Adicionado um condicional `AND date >= OLD.date` em UPDATE
-- Função para calcular total_quantity para um grupo


CREATE OR REPLACE FUNCTION public.calculate_total_quantity_for_group()
RETURNS trigger
LANGUAGE plpgsql
AS $function$
BEGIN
    
    -- STEP 1
    -- Recalcula o grupo ANTIGO (somente em UPDATE)
    
    IF TG_OP = 'UPDATE' THEN
        UPDATE public.positions p
        SET total_quantity = sub.cumulative_sum
        FROM (
            SELECT 
                id,
                SUM(variation) OVER (
                    PARTITION BY holder_id, asset_id, lot_id, financial_account_id
                    ORDER BY date ASC, id ASC
                    ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
                ) AS cumulative_sum
            FROM public.positions
            WHERE holder_id = OLD.holder_id
              AND asset_id = OLD.asset_id
              AND lot_id = OLD.lot_id
              AND financial_account_id = OLD.financial_account_id
        ) sub
        WHERE p.id = sub.id
          AND p.holder_id = OLD.holder_id
          AND p.asset_id = OLD.asset_id
          AND p.lot_id = OLD.lot_id
          AND p.financial_account_id = OLD.financial_account_id
		      AND date >= OLD.date;
    END IF;

    
    -- STEP 2
    -- Recalcula o grupo NOVO (INSERT ou UPDATE)
    
    UPDATE public.positions p
    SET total_quantity = sub.cumulative_sum
    FROM (
        SELECT 
            id,
            SUM(variation) OVER (
                PARTITION BY holder_id, asset_id, lot_id, financial_account_id
                ORDER BY date ASC, id ASC
                ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
            ) AS cumulative_sum
        FROM public.positions
        WHERE holder_id = NEW.holder_id
          AND asset_id = NEW.asset_id
          AND lot_id = NEW.lot_id
          AND financial_account_id = NEW.financial_account_id
    ) sub
    WHERE p.id = sub.id
      AND p.holder_id = NEW.holder_id
      AND p.asset_id = NEW.asset_id
      AND p.lot_id = NEW.lot_id
      AND p.financial_account_id = NEW.financial_account_id
	    AND date >= NEW.date;

    RETURN NULL;
END;
$function$;
